--
-- Create model MaritimeZone
--
CREATE TABLE "maritime_zones" ("id" bigint NOT NULL PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY, "name" varchar(255) NOT NULL, "marine_regions_id" integer NOT NULL UNIQUE CHECK ("marine_regions_id" >= 0), "area_km2" numeric(14, 2) NULL, "source" varchar(100) NOT NULL, "source_version" varchar(50) NOT NULL, "geometry" geometry(MULTIPOLYGON,4326) NOT NULL, "created_at" timestamp with time zone NOT NULL, "updated_at" timestamp with time zone NOT NULL);
--
-- Create model MaritimeCell
--
CREATE TABLE "maritime_cells" ("id" bigint NOT NULL PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY, "latitude" numeric(5, 2) NOT NULL, "longitude" numeric(6, 2) NOT NULL, "center" geometry(POINT,4326) NOT NULL, "created_at" timestamp with time zone NOT NULL, CONSTRAINT "unique_maritime_cell_coordinates" UNIQUE ("latitude", "longitude"));

-- Create model AISObservation
--
CREATE TABLE "ais_observations" ("id" bigint NOT NULL PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY, "period" date NOT NULL, "flag" varchar(3) NOT NULL, "hours" integer NOT NULL CHECK ("hours" >= 0), "vessel_ids" integer NOT NULL CHECK ("vessel_ids" >= 0), "category" varchar(30) NOT NULL, "created_at" timestamp with time zone NOT NULL, "cell_id" bigint NOT NULL, CONSTRAINT "unique_ais_period_flag_cell_category" UNIQUE ("period", "flag", "cell_id", "category"));
--
-- Create model SARObservation
--
